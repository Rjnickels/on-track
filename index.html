function build(){ const d=getData(); const days=new Date(now().getFullYear(),now().getMonth()+1,0).getDate(); // Pre-calc total spent per day const spentByDay = {}; for(let i=1;i<=days;i++){ spentByDay[i] = d.history .filter(h=>h.day===i) .reduce((s,h)=>s+h.amount,0); } const map={}; for(let i=1;i<=days;i++){ // remaining budget AFTER all past spending up to THIS day let remaining = d.budget; for(let j=1;j<=i;j++){ remaining -= (spentByDay[j] || 0); } const daysLeft = days - i + 1; // THIS is the key change: // evenly distribute remaining from this point onward const dailySafe = remaining / daysLeft; map[i] = { remaining, dailySafe }; }
